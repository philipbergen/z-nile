--!strict
local Maze = require("@logic/Maze")
local Awareness = require("@logic/Awareness")
local m = Maze.layout("classic")
local t0 = os.clock()
local n = 0
for i = 1, 2000 do
	local ind = Awareness.indicators(m, { x = 13, y = 23, yaw = 0 }, {
		{ x = 13, y = 20, look = "normal" }, { x = 20, y = 23, look = "normal" }, { x = 6, y = 23, look = "frightened" }, { x = 13, y = 14, look = "normal" },
	})
	n += #ind
end
print("indicators x2000", os.clock() - t0, n / 2000)
local t1 = os.clock()
for i = 1, 2000 do Awareness.radar(m, { x = 13, y = 23, yaw = 0.3 }, {}) end
print("radar x2000", os.clock() - t1)
print(Awareness.pathDistance(m, {x=13,y=23},{x=13,y=11}, 30))
print(Awareness.edgePoint(math.pi, 16/9))
print(Awareness.edgePoint(math.pi/2, 16/9))
