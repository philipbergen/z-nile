--!strict
-- ID AC3 (end to end): scripts/runtime-test against a local fake of Roblox's Luau Execution API. The
-- fake "runs" a script by looking at its text; the command's exit code and report are what is checked.
local fs = require("@lune/fs")
local net = require("@lune/net")
local process = require("@lune/process")
local serde = require("@lune/serde")
local t = require("@tests/lib/t")

-- ID override: pickman's home, where its specs run (its scripts, project files and .cache are there)
local root = (string.gsub(process.cwd, "/+$", ""))
-- ID override: the deploy scripts moved to portal/scripts (pickman's @scripts alias points there too)
local script = `{root}/../portal/scripts/runtime-test`

-- ID override (design review 001 F2): Lune's `env` adds to what the child inherits, so every place variable of the
-- manifest a case does not set is given as "", and a caller's exported ROBLOX_BETA_*_PLACE_ID never reaches the command
local PLACE_VARS = {
	"ROBLOX_BETA_PICKMAN_PLACE_ID",
	"ROBLOX_BETA_GAME_PLACE_ID",
	"ROBLOX_BETA_SNAKE_PLACE_ID",
	"ROBLOX_BETA_INVADERS_PLACE_ID",
	"ROBLOX_BETA_ASTEROIDS_PLACE_ID",
}
local function hermetic(env: { [string]: string }): { [string]: string }
	local out: { [string]: string } = {}
	for _, var in PLACE_VARS do
		out[var] = ""
	end
	for k, v in env do
		out[k] = v
	end
	return out
end
local work = `{root}/.cache/runner-e2e`
local KEY = "KEY-NOT-IN-OUTPUT"

-- ID override: the deploy scripts read experience.json from the current folder (CI runs them from portal/): a fixture
-- manifest of the five places, whose start place (Portal, ROBLOX_BETA_PLACE_ID) builds the fixture's default.project.json
local function manifest(): string
	local places = {}
	for _, p in
		{
			{ "Pickman", "ROBLOX_BETA_PICKMAN_PLACE_ID", "pickman" },
			{ "Snake", "ROBLOX_BETA_SNAKE_PLACE_ID", "snake" },
			{ "Invaders", "ROBLOX_BETA_INVADERS_PLACE_ID", "invaders" },
			{ "Asteroids", "ROBLOX_BETA_ASTEROIDS_PLACE_ID", "asteroids" },
			{ "Portal", "ROBLOX_BETA_PLACE_ID", "portal" },
		}
	do
		table.insert(places, { name = p[1], env = { p[2] }, build = p[3], output = p[3] .. "_version" })
	end
	local builds = {}
	for _, key in { "pickman", "snake", "invaders", "asteroids", "portal" } do
		builds[key] = { project = if key == "portal" then "default.project.json" else `{key}.project.json` }
	end
	return serde.encode("json" :: "json", { places = places, builds = builds })
end

-- a fake: creating a task answers with a path, and polling it completes at once with PASS when the script
-- text has "PASS-ME" in it, else with "nope"
local function serve(): (number, { string }, () -> ())
	local calls: { string } = {}
	local scriptsByTask: { [string]: string } = {}
	for _ = 1, 30 do
		local port = math.random(30000, 60000)
		local ok, handle = pcall(
			net.serve,
			port,
			function(req: any): any
				table.insert(calls, `{req.method} {req.path}`)
				local prefix = "/cloud/v2/universes/111/places/222/versions/5/luau-execution-session-tasks"
				if req.method == "POST" and req.path == prefix then
					local n = tostring(#calls)
					scriptsByTask[n] = (serde.decode("json" :: "json", req.body) :: any).script
					return {
						status = 200,
						body = serde.encode("json" :: "json", { path = `tasks/{n}`, state = "PROCESSING" }),
					}
				end
				local n = string.match(req.path, "^/cloud/v2/tasks/(%d+)$")
				if req.method == "GET" and n then
					local value = if string.find(scriptsByTask[n] or "", "PASS-ME", 1, true)
						then "PASS"
						else "nope"
					return {
						status = 200,
						body = serde.encode(
							"json" :: "json",
							{ path = `tasks/{n}`, state = "COMPLETE", output = { results = { value } } }
						),
					}
				end
				return { status = 404, body = "unexpected" }
			end :: any
		)
		if ok then
			return port, calls, function()
				(handle :: any):stop()
			end
		end
	end
	error("no free port")
end

-- ID override: runs with the place named (Portal: the start place, whose variable is ROBLOX_BETA_PLACE_ID) in `work`,
-- where the fixture manifest is; `extra` env and `cwd` let a case run the real manifest's Pickman place from portal/
local function run(port: number, args: { string }, cwd: string?, extra: { [string]: string }?)
	local argv = { "run", script, "--" }
	if extra == nil then
		table.insert(argv, "--place")
		table.insert(argv, "Portal")
	else
		table.insert(argv, "--place")
		table.insert(argv, "Pickman")
	end
	for _, a in args do
		table.insert(argv, a)
	end
	local env: { [string]: string } = {
		ROBLOX_API_KEY = KEY,
		ROBLOX_BETA_UNIVERSE_ID = "111",
		ROBLOX_BETA_PLACE_ID = "222",
		OPENCLOUD_BASE = `http://127.0.0.1:{port}`,
	}
	if extra then
		for k, v in extra do
			env[k] = v
		end
	end
	return process.exec("lune", argv, { cwd = cwd or work, env = hermetic(env) })
end

return t.spec("ID scripts/runtime-test end to end", function(it)
	fs.writeDir(work)
	fs.writeFile(`{work}/experience.json`, manifest()) -- ID override
	fs.writeFile(`{work}/good.luau`, 'return "PASS-ME"')
	fs.writeFile(`{work}/bad.luau`, 'return "something else"')

	it("all scripts passing: exit 0 and a report", function()
		local port, calls, stop = serve()
		local r = run(port, { "5", `{work}/good.luau` })
		stop()
		t.expect(r.ok).toBe(true)
		t.expect(string.find(r.stdout, `PASS {work}/good.luau`, 1, true) ~= nil).toBe(true)
		t.expect(string.find(r.stdout, "1 passed, 0 failed", 1, true) ~= nil).toBe(true)
		t.expect(#calls >= 2).toBe(true)
		t.expect((string.find(r.stdout .. r.stderr, KEY, 1, true))).toBe(nil)
	end)

	it("one script that doesn't pass fails the command, names it, and the others still run", function()
		local port, _, stop = serve()
		local r = run(port, { "5", `{work}/good.luau`, `{work}/bad.luau`, `{work}/good.luau` })
		stop()
		t.expect(r.ok).toBe(false)
		t.expect(string.find(r.stdout, `FAIL {work}/bad.luau: returned nope`, 1, true) ~= nil).toBe(true)
		t.expect(string.find(r.stdout, "2 passed, 1 failed", 1, true) ~= nil).toBe(true)
	end)

	it(
		"with no files it runs everything under the place's runtime folder (Pickman's tests/runtime)",
		function()
			local port, _, stop = serve()
			-- ID override: run from portal/ (the real experience.json) as CI does; Pickman gets the id the fake knows and
			-- the start place another (two places with one id are refused); the scripts are named relative to the manifest
			local r = run(
				port,
				{ "5" },
				`{root}/../portal`,
				{ ROBLOX_BETA_PICKMAN_PLACE_ID = "222", ROBLOX_BETA_PLACE_ID = "333" }
			)
			stop()
			-- the checked-in structure script has no PASS-ME marker, so against this fake it fails, by name
			t.expect(string.find(r.stdout, "FAIL ../pickman/tests/runtime/structure.luau", 1, true) ~= nil)
				.toBe(true)
		end
	)

	it("the version is part of the request: a different one is not found", function()
		local port, _, stop = serve()
		local r = run(port, { "6", `{work}/good.luau` })
		stop()
		t.expect(r.ok).toBe(false) -- the fake only knows version 5
	end)
end)
