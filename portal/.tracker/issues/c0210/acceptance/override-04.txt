--!strict
-- ID AC3 (end to end): scripts/deploy-beta against a local fake of Roblox. The real command runs with the
-- real Rojo on a tiny fixture project, and what reaches the fake server, what the command prints and what it
-- appends to $GITHUB_OUTPUT is checked. This is the glue the unit specs of OpenCloud can't see.
local fs = require("@lune/fs")
local net = require("@lune/net")
local process = require("@lune/process")
local serde = require("@lune/serde")
local t = require("@tests/lib/t")

-- ID override: pickman's home, where its specs run (its scripts, project files and .cache are there)
local root = (string.gsub(process.cwd, "/+$", ""))
-- ID override: the deploy scripts moved to portal/scripts (pickman's @scripts alias points there too)
local script = `{root}/../portal/scripts/deploy-beta`

-- ID override (design review 001 F2): Lune's `env` adds to what the child inherits, so every place variable of the
-- manifest a case does not set is given as "", and a caller's exported ROBLOX_BETA_*_PLACE_ID never reaches the command
local PLACE_VARS = {
	"ROBLOX_BETA_PICKMAN_PLACE_ID",
	"ROBLOX_BETA_GAME_PLACE_ID",
	"ROBLOX_BETA_SNAKE_PLACE_ID",
	"ROBLOX_BETA_INVADERS_PLACE_ID",
	"ROBLOX_BETA_ASTEROIDS_PLACE_ID",
}
local function hermetic(env: { [string]: string }): { [string]: string }
	local out: { [string]: string } = {}
	for _, var in PLACE_VARS do
		out[var] = ""
	end
	for k, v in env do
		out[k] = v
	end
	return out
end
local work = `{root}/.cache/deploy-e2e`
local KEY = "KEY-NOT-IN-OUTPUT"

-- ID override: the deploy scripts read experience.json from the current folder (CI runs them from portal/): a fixture
-- manifest of the five places, whose start place (Portal, ROBLOX_BETA_PLACE_ID) builds the fixture's default.project.json
-- and reports it as `version` (so the one publish below is the old one)
local function manifest(): string
	local places = {}
	for _, p in
		{
			{ "Pickman", "ROBLOX_BETA_PICKMAN_PLACE_ID", "pickman" },
			{ "Snake", "ROBLOX_BETA_SNAKE_PLACE_ID", "snake" },
			{ "Invaders", "ROBLOX_BETA_INVADERS_PLACE_ID", "invaders" },
			{ "Asteroids", "ROBLOX_BETA_ASTEROIDS_PLACE_ID", "asteroids" },
			{ "Portal", "ROBLOX_BETA_PLACE_ID", "portal" },
		}
	do
		table.insert(places, {
			name = p[1],
			env = { p[2] },
			build = p[3],
			output = if p[1] == "Portal" then "version" else p[3] .. "_version",
		})
	end
	local builds = {}
	for _, key in { "pickman", "snake", "invaders", "asteroids", "portal" } do
		builds[key] = { project = if key == "portal" then "default.project.json" else `{key}.project.json` }
	end
	return serde.encode("json" :: "json", { places = places, builds = builds })
end

type Seen = {
	method: string,
	path: string,
	query: { [string]: string },
	key: string?,
	contentType: string?,
	body: string,
}

-- a fake Roblox answering every request with (status, body); `seen` collects the requests
local function serve(status: number, body: string): (number, { Seen }, () -> ())
	local seen: { Seen } = {}
	for _ = 1, 30 do
		local port = math.random(30000, 60000)
		local ok, handle = pcall(
			net.serve,
			port,
			function(req: any): any
				table.insert(seen, {
					method = req.method,
					path = req.path,
					query = req.query,
					key = req.headers["x-api-key"],
					contentType = req.headers["content-type"],
					body = req.body or "",
				})
				return { status = status, body = body }
			end :: any
		)
		if ok then
			return port, seen, function()
				(handle :: any):stop()
			end
		end
	end
	error("no free port")
end

local function run(port: number, extra: { [string]: string }?)
	local env: { [string]: string } = {
		ROBLOX_API_KEY = KEY,
		ROBLOX_BETA_UNIVERSE_ID = "111",
		ROBLOX_BETA_PLACE_ID = "222",
		OPENCLOUD_BASE = `http://127.0.0.1:{port}`,
		GITHUB_OUTPUT = "",
	}
	if extra then
		for k, v in extra do
			env[k] = v
		end
	end
	return process.exec("lune", { "run", script }, { env = hermetic(env), cwd = work })
end

return t.spec("ID scripts/deploy-beta end to end", function(it)
	fs.writeDir(work)
	fs.writeFile(
		`{work}/default.project.json`,
		'{"name":"fixture","tree":{"$className":"DataModel","ReplicatedStorage":{"$className":"ReplicatedStorage"}}}'
	)
	fs.writeFile(`{work}/experience.json`, manifest()) -- ID override
	local out = `{work}/github-output`

	it("builds the place and publishes it with the right request, then reports the version", function()
		fs.writeFile(out, "existing=1\n")
		local port, seen, stop = serve(200, '{"versionNumber":42}')
		local r = run(port, { GITHUB_OUTPUT = out })
		stop()
		t.expect(r.ok).toBe(true)
		t.expect(#seen).toBe(1)
		t.expect(seen[1].method).toBe("POST")
		t.expect(seen[1].path).toBe("/universes/v1/111/places/222/versions")
		t.expect(seen[1].query.versionType).toBe("Published")
		t.expect(seen[1].key).toBe(KEY)
		t.expect(seen[1].contentType).toBe("application/octet-stream")
		t.expect(string.sub(seen[1].body, 1, 8)).toBe("<roblox!") -- a real built place, not text
		t.expect(string.find(r.stdout, "version=42", 1, true) ~= nil).toBe(true)
		t.expect((string.find(r.stdout .. r.stderr, KEY, 1, true))).toBe(nil)
		t.expect(fs.readFile(out)).toBe("existing=1\nversion=42\n") -- appended, nothing lost
	end)

	it("without GITHUB_OUTPUT it still prints the version and writes no file", function()
		local port, _, stop = serve(200, '{"versionNumber":7}')
		local r = run(port)
		stop()
		t.expect(r.ok).toBe(true)
		t.expect(string.find(r.stdout, "version=7", 1, true) ~= nil).toBe(true)
	end)

	it("a refusal fails the command, shows the status and body, and reports no version", function()
		fs.writeFile(out, "")
		local port, seen, stop = serve(403, '{"errors":[{"code":9009,"message":"WAF Blocked Request"}]}')
		local r = run(port, { GITHUB_OUTPUT = out })
		stop()
		t.expect(r.ok).toBe(false)
		t.expect(#seen).toBe(1) -- not retried
		t.expect(string.find(r.stdout, "403", 1, true) ~= nil).toBe(true)
		t.expect(string.find(r.stdout, "WAF Blocked Request", 1, true) ~= nil).toBe(true)
		t.expect((string.find(r.stdout, "version=", 1, true))).toBe(nil)
		t.expect((string.find(r.stdout .. r.stderr, KEY, 1, true))).toBe(nil)
		t.expect(fs.readFile(out)).toBe("")
	end)

	it("a 200 without a version fails and writes nothing", function()
		fs.writeFile(out, "")
		local port, _, stop = serve(200, "{}")
		local r = run(port, { GITHUB_OUTPUT = out })
		stop()
		t.expect(r.ok).toBe(false)
		t.expect(fs.readFile(out)).toBe("")
	end)

	it("a base that is not loopback is refused before anything is built or sent", function()
		local r = run(1, { OPENCLOUD_BASE = "https://evil.example" })
		t.expect(r.ok).toBe(false)
		t.expect(string.find(r.stdout .. r.stderr, "loopback", 1, true) ~= nil).toBe(true)
		t.expect((string.find(r.stdout .. r.stderr, KEY, 1, true))).toBe(nil)
	end)
end)
