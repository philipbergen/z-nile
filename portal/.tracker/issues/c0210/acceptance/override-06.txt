--!strict
-- ID AC1: the publish plan, as pure data. ID override: the places come from a manifest (here the two places of
-- pickman as ID made them: the game and, at the start place, the lobby). Without the game place's variable the
-- game is not published and the lobby alone goes to the start place (ID: a place whose variable is unset is
-- skipped); with it, the game goes to its own place and the lobby, which is told the game place's id, to the start
-- place. A bad id fails, naming the variable and never the key.
local Beta = require("@scripts/lib/beta")
local Experience = require("@scripts/lib/experience")
local t = require("@tests/lib/t")

local MANIFEST = [[
{
  "places": [
    { "name": "Pickman", "env": ["ROBLOX_BETA_PICKMAN_PLACE_ID", "ROBLOX_BETA_GAME_PLACE_ID"], "build": "game", "output": "version" },
    { "name": "Snake", "env": ["ROBLOX_BETA_SNAKE_PLACE_ID"], "build": "snake", "output": "snake_version" },
    { "name": "Invaders", "env": ["ROBLOX_BETA_INVADERS_PLACE_ID"], "build": "invaders", "output": "invaders_version" },
    { "name": "Asteroids", "env": ["ROBLOX_BETA_ASTEROIDS_PLACE_ID"], "build": "asteroids", "output": "asteroids_version" },
    { "name": "Portal", "env": ["ROBLOX_BETA_PLACE_ID"], "build": "lobby", "output": "lobby_version" }
  ],
  "builds": {
    "game": { "project": "default.project.json", "runtime": "tests/runtime", "attributes": { "LobbyPlaceId": "Portal" } },
    "lobby": { "project": "lobby.project.json", "runtime": "tests/runtime/lobby", "attributes": { "GamePlaceId": "Pickman" } },
    "snake": { "project": "s.project.json" },
    "invaders": { "project": "i.project.json" },
    "asteroids": { "project": "a.project.json" }
  }
}
]]
local manifest = Experience.parse(MANIFEST) :: Experience.Manifest

local BASE: { [string]: string? } = {
	ROBLOX_API_KEY = "KEY-NOT-IN-OUTPUT",
	ROBLOX_BETA_UNIVERSE_ID = "10768553777",
	ROBLOX_BETA_PLACE_ID = "82331952549326",
}

local function env(game: string?): { [string]: string? }
	local e: { [string]: string? } = table.clone(BASE)
	e.ROBLOX_BETA_GAME_PLACE_ID = game
	return e
end

local function plan(game: string?): Beta.Plan
	local p = Beta.plan(env(game), manifest)
	assert(p, "configured, so a plan")
	return p
end

-- the game target; it is told the lobby's id as LobbyPlaceId (ID's attribute), never a GamePlaceId
local function expectGame(target: Beta.Target, placeId: string, lobby: number)
	t.expect(target.name).toBe("Pickman")
	t.expect(target.project).toBe("default.project.json")
	t.expect(target.placeId).toBe(placeId)
	t.expect(target.output).toBe("version")
	t.expect(target.runtime).toBe("tests/runtime")
	t.expect(target.attributes.LobbyPlaceId).toBe(lobby)
	t.expect(target.attributes.GamePlaceId).toBe(nil)
end

local function throws(game: string, needle: string)
	local ok, err = pcall(Beta.plan, env(game), manifest)
	t.expect(ok).toBe(false)
	t.expect(string.find(tostring(err), needle, 1, true) ~= nil).toBe(true)
	t.expect((string.find(tostring(err), "KEY-NOT-IN-OUTPUT", 1, true))).toBe(nil)
end

return t.spec("ID Beta.plan", function(it)
	it("names the game place's variable", function()
		t.expect(manifest.places[1].env[2]).toBe("ROBLOX_BETA_GAME_PLACE_ID")
	end)

	it("not configured is nil, whatever the game place's variable says", function()
		for _, name in { "ROBLOX_API_KEY", "ROBLOX_BETA_UNIVERSE_ID", "ROBLOX_BETA_PLACE_ID" } do
			local e = env("4815162342")
			e[name] = ""
			t.expect(Beta.plan(e, manifest)).toBe(nil)
			e[name] = nil
			t.expect(Beta.plan(e, manifest)).toBe(nil)
		end
	end)

	it(
		"without the game place's id: the game is not published, and the lobby alone goes to the start place (ID)",
		function()
			for _, game in { "UNSET", "" } do -- a nil in a table literal is skipped, so a sentinel stands for unset
				local p = plan(if game == "UNSET" then nil else game)
				t.expect(#p.targets).toBe(1)
				t.expect(p.targets[1].name).toBe("Portal")
				t.expect(p.targets[1].project).toBe("lobby.project.json")
				t.expect(p.targets[1].attributes.GamePlaceId).toBe(0) -- no game place to go to
				t.expect(string.find(p.skipped[1], "ROBLOX_BETA_PICKMAN_PLACE_ID", 1, true) ~= nil).toBe(true)
			end
		end
	)

	it(
		"with the game place's id: the game to it first, then the lobby to the start place, told the id",
		function()
			local p = plan("4815162342")
			t.expect(#p.targets).toBe(2)
			expectGame(p.targets[1], "4815162342", 82331952549326)
			local lobby = p.targets[2]
			t.expect(lobby.name).toBe("Portal")
			t.expect(lobby.project).toBe("lobby.project.json")
			t.expect(lobby.placeId).toBe("82331952549326")
			t.expect(lobby.attributes.GamePlaceId).toBe(4815162342)
			t.expect(lobby.attributes.LobbyPlaceId).toBe(nil)
			t.expect(lobby.output).toBe("lobby_version")
			t.expect(lobby.runtime).toBe("tests/runtime/lobby")
		end
	)

	it("the lobby is told exactly the id it is given, as a number, up to 2^53-1", function()
		for _, game in { "1", "7", "123456789012345", "9007199254740991" } do
			local lobby = Beta.target(plan(game), "Portal")
			assert(lobby, "split, so a lobby")
			t.expect(lobby.attributes.GamePlaceId).toBe(tonumber(game))
			t.expect(type(lobby.attributes.GamePlaceId)).toBe("number")
			t.expect((Beta.target(plan(game), "Pickman") :: Beta.Target).placeId).toBe(game)
		end
	end)

	it("a game place id that is not a place id fails, naming the variable", function()
		for _, bad in
			{ "0", "007", "-5", "1.5", "12a", "1/../2", " 12", "0x10", "1e9", "9007199254740992", "１２" }
		do
			throws(bad, "ROBLOX_BETA_GAME_PLACE_ID")
		end
	end)

	it("a game place that is the start place fails: the lobby would overwrite the game", function()
		throws("82331952549326", "ROBLOX_BETA_GAME_PLACE_ID")
	end)

	it("a bad start place id still fails as before, naming its variable", function()
		local e = env("4815162342")
		e.ROBLOX_BETA_PLACE_ID = "12a"
		local ok, err = pcall(Beta.plan, e, manifest)
		t.expect(ok).toBe(false)
		t.expect(string.find(tostring(err), "ROBLOX_BETA_PLACE_ID", 1, true) ~= nil).toBe(true)
	end)

	it("target finds a place in the plan, and nil for one it does not publish", function()
		t.expect(Beta.target(plan(nil), "Pickman")).toBe(nil)
		t.expect((Beta.target(plan(nil), "Portal") :: Beta.Target).placeId).toBe("82331952549326")
		t.expect((Beta.target(plan("4815162342"), "Pickman") :: Beta.Target).placeId).toBe("4815162342")
	end)

	it("never changes the environment it reads", function()
		local e = env("4815162342")
		local before = table.clone(e)
		Beta.plan(e, manifest)
		t.expect(e).toEqual(before)
	end)
end)
