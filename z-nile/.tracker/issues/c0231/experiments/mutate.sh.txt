#!/usr/bin/env bash
# c0231 mutation check (REWRITTEN after design review 001 F2; written, not run, 2026-11-15). Applies each mutant to the
# prototype inside a throwaway COPY of the z-nile home under <checkout>/.cache, never in the checkout, runs the row's
# spec there and counts a kill only on a real assertion failure. Run from a worktree root, bounded:
#   ./zn run --time 600 -- bash <this file>
# The fixture needs the prototype applied: experiments/prototype.patch is applied to the copy (never to the checkout).
set -u
ROOT=$(git rev-parse --show-toplevel) || exit 2
cd "$ROOT" || exit 2
HOME_DIR=
for d in z-nile/.tracker/issues/c0231 z-nile/.tracker/issues/active/c0231; do [[ -d $d/acceptance ]] && HOME_DIR=$d && break; done
if [[ -z $HOME_DIR && -d z-nile/tests/acceptance/c0231 ]]; then HOME_DIR=z-nile/tests/acceptance/c0231; fi
[[ -n $HOME_DIR ]] || { echo "NO ACCEPTANCE FOLDER FOUND"; exit 2; }
FIX=$(mktemp -d "$ROOT/.cache/c0231-mutants-XXXXXX") || exit 2
trap 'rm -rf "$FIX"' EXIT
cp -R z-nile "$FIX/z-nile"
cp "$ROOT/.luaurc" "$FIX/.luaurc" 2>/dev/null
mkdir -p "$FIX/acc"
cp "$HOME_DIR"/acceptance/*.luau "$FIX/acc/" 2>/dev/null || cp "$HOME_DIR"/*.luau "$FIX/acc/"
PATCH=$ROOT/z-nile/.tracker/issues/c0231/experiments/prototype.patch
[[ -f $PATCH ]] || PATCH=$(find "$ROOT/z-nile" -path '*c0231/experiments/prototype.patch' | head -1)
(cd "$FIX" && git init -q . && git apply "$PATCH") || { echo "PROTOTYPE DOES NOT APPLY"; exit 2; }
F=$FIX/z-nile/engine/boundcmd.luau
BACKUP=$(mktemp "$FIX/backup-XXXXXX")
cp "$F" "$BACKUP"

run() { # <id> <spec> <perl substitution>
	local id=$1 spec=$2 sub=$3
	cp "$BACKUP" "$F"
	perl -0pi -e "$sub" "$F"
	if cmp -s "$BACKUP" "$F"; then echo "$id NOT APPLIED"; return; fi
	local out
	out=$(cd "$FIX/z-nile" && lune run tests/run -- "$FIX/acc/$spec" 2>&1)
	if printf "%s" "$out" | grep -qiE "syntax|could not load|attempt to|unknown require"; then
		echo "$id NO RUN ($spec): load or syntax error"
	elif printf "%s" "$out" | grep -qE "[1-9][0-9]* failed"; then
		echo "$id killed ($spec): $(printf "%s" "$out" | grep -m1 -E "expected" | cut -c1-140)"
	else
		echo "$id SURVIVED ($spec)"
	fi
}
run M1-no-rule refused.spec.luau 's/name == "yes"\n/name == "no-such-program"\n/'
run M2-raw-name refused.spec.luau 's/name == "yes"\n/program == "yes"\n/'
run M3-any-pipe consumers.spec.luau 's/if program == nil or basename\(program\) ~= "head" then/if program == nil then/'
run M4-last-stage consumers.spec.luau 's/pipe = if piped then commands\[k \+ 1\] else nil/pipe = if piped then commands[#commands] else nil/'
run M5-negative consumers.spec.luau 's/\(value ~= nil and string.sub\(value, 1, 1\) == "-"\)/false/'
run M6-xargs consumers.spec.luau 's/if basename\(words\[k\]\) == "xargs" then/if false then/'
run M7-redirect-ignored refused.spec.luau 's/\n\s*and not stdoutAway\(args\)//'
run M8-any-redirect allowed.spec.luau 's/string.match\(a, "\^1\?>"\)/string.match(a, "^%d*>")/'
run M9-refuse-all allowed.spec.luau 's/and not \(\n(\s*)after ~= nil/and not (\n$1false/'
run M10-no-case mentions.spec.luau 's/pattern = cases > 0 and ends\[k\] == "\)"/pattern = false/'
run M11-case-never-closes mentions.spec.luau 's/elseif program == "esac" and cases > 0 then\n(\s*)cases -= 1/elseif program == "esac" and cases > 0 then\n$1cases += 0/'
run M12-any-paren refused.spec.luau 's/pattern = cases > 0 and ends\[k\] == "\)"/pattern = ends[k] == ")"/'
run M13-monitor contexts.spec.luau 's/name == "yes"\n(\s*)and not CTX.monitor/name == "yes"\n$1and true/'
run M14-caffeinate-drops allowed.spec.luau 's/checkCommand\(rest, depth, nil, after\)/checkCommand(rest, depth)/'
run M15-rule-name refused.spec.luau 's/return refuse\(\n(\s*)"follow",\n(\s*)name,\n(\s*)line,\n(\s*)`\\`yes\\`/return refuse(\n$1"unbounded",\n$2name,\n$3line,\n$4`\\`yes\\`/'
run M16-no-head-hint refusal.spec.luau 's/\(\\`yes \| head -3\\`\)/(head)/'
run M17-pipe-only allowed.spec.luau 's/local piped = ends\[k\] == "\|" or ends\[k\] == "\|&"/local piped = ends[k] == "|"/'
run M18-exempt-args refused.spec.luau 's/name == "yes"\n/name == "yes" and not EXEMPT_ARG[args[1] or ""]\n/'
run M19-bounded-stage-ignored allowed.spec.luau 's/ or boundedish\(after.pipe\)//'
# review 001 F8 (written, not run): rows added
run M20-head-raw-name allowed.spec.luau 's/if program == nil or basename\(program\) ~= "head" then/if basename(words[1]) ~= "head" then/'
run M21-no-next-stage refused.spec.luau 's/pipe = if piped then commands\[k \+ 1\] else nil/pipe = if piped then (commands[k + 1] or {"head"}) else nil/'
run M22-gt-only refused.spec.luau 's/string.match\(a, "\^&>"\)/false/'
