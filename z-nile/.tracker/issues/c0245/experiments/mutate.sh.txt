#!/usr/bin/env bash
# Runs each mutant in the mutation clone against c0245's whole check.sh and reports which criteria fail, and whether
# each failure is an assertion (not a load or syntax error). Restores the files after each mutant.
M=$1
S=$2
cd "$M" || exit 1
for name in $(python3 -I "$S/mutants.py" list); do
  cp z-nile/engine/boundcmd.luau "$S/b.orig"; cp z-nile/guard-bash.luau "$S/g.orig"
  cp z-nile/tests/acceptance/c0191/boundcmd.spec.luau "$S/s.orig"; cp z-nile/skills/zz-test/SKILL.md "$S/t.orig"
  if ! python3 -I "$S/mutants.py" "$M" "$name" >/dev/null; then
    echo "$name: DID NOT APPLY"
  else
    out=$(bash z-nile/.tracker/issues/c0245/check.sh 2>&1)
    failed=$(printf '%s\n' "$out" | grep -E '^FAIL  AC' | sed -E 's/^FAIL  (AC[0-9]+):.*/\1/' | tr '\n' ' ')
    loads=$(printf '%s\n' "$out" | grep -cE 'could not load|syntax error|attempt to call|error requiring')
    echo "$name: failed [${failed}] load-errors=$loads"
  fi
  cp "$S/b.orig" z-nile/engine/boundcmd.luau; cp "$S/g.orig" z-nile/guard-bash.luau
  cp "$S/s.orig" z-nile/tests/acceptance/c0191/boundcmd.spec.luau; cp "$S/t.orig" z-nile/skills/zz-test/SKILL.md
done
